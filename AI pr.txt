## 1. 8-Puzzle Problem using A\* Search

CODE
import heapq

goal = (1,2,3,4,5,6,7,8,0)

def h(s):
    return sum(abs(i//3-s.index(x)//3)+abs(i%3-s.index(x)%3)
               for i,x in enumerate(s) if x)

def astar(start):
    q = [(h(start), 0, start, [])]
    seen = set()
    while q:
        f,g,s,p = heapq.heappop(q)
        if s in seen: continue
        seen.add(s)
        p = p+[s]
        if s == goal: return p
        z = s.index(0)
        for d in [-3,3,-1,1]:
            n = z+d
            if 0 <= n < 9 and not (d==-1 and z%3==0) and not (d==1 and z%3==2):
                t = list(s)
                t[z],t[n] = t[n],t[z]
                t = tuple(t)
                if t not in seen:
                    heapq.heappush(q,(g+1+h(t),g+1,t,p))
    return None

s = tuple(map(int,input("Enter 9 numbers: ").split()))
p = astar(s)
if p:
    for x in p:
        print(x[:3],x[3:6],x[6:])
    print("Moves:",len(p)-1)
else:
    print("No solution")

## 2. Adversarial Search using Alpha-Beta Pruning


CODE
def minimax(d,i,maxp,a,b,v):
    if d == 3:
        return v[i]

    if maxp:
        best = -999
        for j in range(2):
            best = max(best,minimax(d+1,i*2+j,False,a,b,v))
            a = max(a,best)
            if b <= a: break
        return best
    else:
        best = 999
        for j in range(2):
            best = min(best,minimax(d+1,i*2+j,True,a,b,v))
            b = min(b,best)
            if b <= a: break
        return best

v = list(map(int,input("Enter 8 values: ").split()))
print("Best value:",minimax(0,0,True,-999,999,v))

## 3. Tower of Hanoi using State Space Search

def hanoi(n,a,b,c):
    if n == 1:
        print(a,"->",c)
        return
    hanoi(n-1,a,c,b)
    print(a,"->",c)
    hanoi(n-1,b,a,c)

def solve(b):
    for r in range(9):
        for c in range(9):
            if b[r][c] == 0:
                for n in range(1,10):
                    if (n not in b[r] and
                        all(b[i][c]!=n for i in range(9)) and

                            for j in range(c//3*3,c//3*3+3))):
                        b[r][c] = n
                        if solve(b): return True
                        b[r][c] = 0
                return False
    return True

## 5. BFS for Robot Path Planning

CODE
from collections import deque

def bfs(g,s,goal):
    q = deque([(s,[s])])
    seen = {s}

    while q:
        (r,c),p = q.popleft()
        if (r,c) == goal: return p

        for dr,dc in [(0,1),(0,-1),(1,0),(-1,0)]:
            x,y = r+dr,c+dc
            if 0<=x<len(g) and 0<=y<len(g[0]) and g[x][y]==0 and (x,y) not in seen:
                seen.add((x,y))
                q.append(((x,y),p+[(x,y)]))
    return None

g = [[0,0,0],[1,1,0],[0,0,0]]
print("Path:",bfs(g,(0,0),(2,2)))

## 6. DFS for Water Jug Problem

CODE
def dfs(a,b,t):
    stack = [(0,0,[])]
    seen = set()

    while stack:
        x,y,p = stack.pop()
        if (x,y) in seen: continue
        seen.add((x,y))
        p = p+[(x,y)]

        if x==t or y==t: return p

        d = min(x,b-y)
        e = min(y,a-x)

        states = [(a,y),(x,b),(0,y),(x,0),
                  (x-d,y+d),(x+e,y-e)]

        for s in states:
            if s not in seen:
                stack.append((*s,p))
    return None

a = int(input("Jug 1 capacity: "))
b = int(input("Jug 2 capacity: "))
t = int(input("Target: "))
print("Solution:",dfs(a,b,t))

```
Jug 1 capacity: 4
Jug 2 capacity: 3
Target: 2
```

Expected Output:

```

(1, 0), (0, 1), (4, 1), (2, 3)]
```

## 7. 8-Queens using Backtracking

N = 8
board = [-1]*N

def solve(r):
    if r == N: return True

    for c in range(N):
        if all(board[i]!=c and abs(board[i]-c)!=abs(i-r)
               for i in range(r)):
            board[r] = c
            if solve(r+1): return True
            board[r] = -1
    return False

if solve(0):
    for c in board:
        print(" ".join("Q" if j==c else "." for j in range(N)))

## 9. Number Guessing Intelligent Agent

CODE
low,high = 1,100

print("Think of a number between 1 and 100")

while low <= high:
    mid = (low+high)//2
    print("AI Guess:",mid)
    s = input("Higher/Lower/Correct: ").lower()

    if s == "correct":
        print("AI guessed correctly!")
        break
    elif s == "higher":
        low = mid+1
    elif s == "lower":
        high = mid-1
    else:
        print("Invalid input")

## 11. Game-Playing Agent using Minimax Algorithm

CODE
def minimax(d, maxp, v):
    if d == 2:
        return v.pop(0)

    if maxp:
        return max(minimax(d+1,False,v),
                   minimax(d+1,False,v))
    else:
        return min(minimax(d+1,True,v),
                   minimax(d+1,True,v))
